    import 'server-only';
    import { db } from '@/lib/db';
    import { employees, vacationRecords } from '@/lib/db/schema';
    import { eq, asc } from 'drizzle-orm';

    export type Employee = typeof employees.$inferSelect;
    export type NewEmployee = typeof employees.$inferInsert;

    export async function getEmployees(): Promise<Employee[]> {
      return db
        .select()
        .from(employees)
        .where(eq(employees.isActive, true))
        .orderBy(asc(employees.fullName));
    }

    export async function addEmployee(data: {
      fullName: string;
      hireDate: string;
      isDeel?: boolean;
      email?: string | null;
    }): Promise<Employee> {
      const [row] = await db.insert(employees).values(data).returning();
      return row;
    }

    export async function updateEmployee(data: {
      id: number;
      fullName?: string;
      hireDate?: string;
      isActive?: boolean;
      isDeel?: boolean;
      email?: string | null;
    }): Promise<Employee> {
      const { id, ...rest } = data;
      const [row] = await db
        .update(employees)
        .set(rest)
        .where(eq(employees.id, id))
        .returning();
      return row;
    }

    export async function deleteEmployee(id: number): Promise<void> {
      // Manual cascade: delete records first, then employee (mirrors Electron behavior)
      await db.delete(vacationRecords).where(eq(vacationRecords.employeeId, id));
      await db.delete(employees).where(eq(employees.id, id));
    }
    ```

    import 'server-only';
    import { db } from '@/lib/db';
    import { vacationRecords } from '@/lib/db/schema';
    import { eq, asc, desc } from 'drizzle-orm';

    export type VacationRecord = typeof vacationRecords.$inferSelect;
    export type NewVacationRecord = typeof vacationRecords.$inferInsert;

    export async function getVacationRecords(employeeId: number): Promise<VacationRecord[]> {
      return db
        .select()
        .from(vacationRecords)
        .where(eq(vacationRecords.employeeId, employeeId))
        .orderBy(desc(vacationRecords.startDate));
    }

    export async function getAllVacationRecords(): Promise<VacationRecord[]> {
      return db
        .select()
        .from(vacationRecords)
        .orderBy(asc(vacationRecords.employeeId), desc(vacationRecords.startDate));
    }

    export async function addVacationRecord(data: NewVacationRecord): Promise<VacationRecord> {
      const [row] = await db.insert(vacationRecords).values(data).returning();
      return row;
    }

    export async function updateVacationRecord(data: {
      id: number;
      recordType?: 'period' | 'days_sum' | 'balance_reset';
      startDate?: string | null;
      endDate?: string | null;
      daysCount?: number | null;
      year?: number | null;
      note?: string | null;
    }): Promise<VacationRecord> {
      const { id, ...rest } = data;
      const [row] = await db
        .update(vacationRecords)
        .set(rest)
        .where(eq(vacationRecords.id, id))
        .returning();
      return row;
    }

    export async function deleteVacationRecord(id: number): Promise<void> {
      await db.delete(vacationRecords).where(eq(vacationRecords.id, id));
    }
    ```
